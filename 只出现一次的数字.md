class Solution {
public:
    int singleNumber(vector<int>& nums)
    {
        unordered_map 
         const int n=3*10e5,m=3*10e4;
         vector<int> once(n);
         for (int i=0;i<size(nums);i++)
         {
             once[nums[i]+m]++;
         }
         for(int j=0;j<n;j++)
         {
             if (once[j]==1)
             {
                 return j-m;
             }
         }
    return 0;
    }
};
