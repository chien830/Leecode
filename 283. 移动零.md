```
class Solution {
public:
    void moveZeroes(vector<int>& nums) {
        int l=0,r=0,n=size(nums);
        while(r<n){
            if(nums[r])
            {
                swap(nums[l],nums[r]);
                l++;
            }
            r++;
        }
    }
};
```
双指针问题
左指针指向当前已经处理好的序列的尾部，右指针指向待处理序列的头部。


```
class Solution {
public:
   void moveZeroes(vector<int>& nums) {
       stable_sort(nums.begin(), nums.end(),[](int a, int b) {return a != 0 && b == 0;});
   }
};
```
