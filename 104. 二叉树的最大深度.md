```
class Solution {
public:
    int maxDepth(TreeNode* root) {
        int maxl=0,maxr=0;
        if (root == nullptr) {
            return 0;
        }
        maxr=maxDepth(root->right);
        maxl=maxDepth(root->left);
        if(maxl>maxr) 
        {
            return maxl+1;
        }
        else return maxr+1;
        
    }
};
```
